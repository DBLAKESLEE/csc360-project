# key
# t0: i
# t1: array[i]
# t2: current value
# t3: j
# t4: array[j]
# t5: value at array[j]

.data
		array: .word 4, 5, -1, 100, 1, 22
		n: .word 6

.text
		lw a1, n
		la a0, array
		li t0, 1

for_loop:
		@C@bge@ @D@t0@, @C@a1, end@						@C@# branch hazard and dependency from line 39@
		slli @D@t1@, t0, 2
		add @D@t1@, a0, @D@t1@ 						@D@# add rs2 depends on rd of slli@
		lw t2, 0(t1)
		addi t3, t0, -1

while_loop:
		@C@blt@ @D@t3@, @C@x0, insert@			@C@# branch hazard and dependency from line 32@
		slli @D@t4@, t3, 2
		add @D@t4@, a0, @D@t4@							@D@# add rs2 depends on rd of slli@
		lw @L@t5@, 0(@D@t4@)									@L@# here there is a load use hazard where lw writes t5@
		@C@ble@ @L@t5@, @C@t2, insert@			@L@# in MEM/WB, but ble needs t5 in EX@
		sw @L@t5@, 4(t4)									@L@# similarly, t5 written in MEM/WB but need as input for sw@
		addi @C@t3@, @D@t3@, -1								
		@C@beq x0, x0, while_loop@

insert:
		slli @D@t4@, t3, 2
		add @D@t4@, a0, @D@t4@							@D@# add rs2 depends on rd of slli@
		sw t2, 4(t4)
		addi @C@t0@, @D@t0@, 1
		@C@beq x0, x0, for_loop@	@C@# branch hazard@
	
end:
		li a7, 10
		ecall

	\end{lstlisting}
